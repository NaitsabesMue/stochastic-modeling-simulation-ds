\section{The question and the method}

\begin{frame}{Why this course? What do you gain?}
  An operations log shows what happened under one configuration. A decision often concerns a configuration never observed.
  \vspace{0.5em}
  \begin{itemize}
    \item Turn a concrete question, such as adding a bank-call agent, into outcomes and assumptions.
    \item Build a small Python simulation of a proposed change.
    \item Quantify simulation uncertainty and find assumptions that fail against real data.
    \item Communicate a decision, including what the available evidence cannot establish.
  \end{itemize}
  \vspace{0.3em}
  \textbf{Your project:} carry that reasoning from observed data to a defensible recommendation.
\end{frame}

\begin{frame}{A question the existing records cannot answer}
  \large
  Would another agent reduce waiting at the bank's call centre?
  \normalsize
  \vspace{0.6em}
  \begin{itemize}
    \item The records describe what happened with the staffing and routing that were in place.
    \item They contain no observation of the proposed staffing change.
    \item A simulation can explore that change, \emph{if} its assumptions survive comparison with observed behaviour.
  \end{itemize}
  \vspace{0.4em}
  \textbf{Today:} start with arrivals. We cannot estimate the effect on waiting from arrival counts alone.
\end{frame}

  \begin{enumerate}

    \item \textbf{Notebook and exercises:} open your own computer; use Python, inspect data and record your reasoning.
    \item \textbf{Final 10 minutes in sessions 1--6:} complete a printed multiple-choice quiz individually, without devices or LLMs.
  \end{enumerate}
  \vspace{0.5em}
  Today the opening diagnostic is part of the screen-free segment. Session 7 ends with the project, without a CC quiz.

\begin{frame}{Assessment: six paper quizzes}
  \begin{itemize}
    \item One graded end-of-session MCQ in each of sessions 1--6, scored out of 20. An absence scores zero; session 7 has no CC quiz.
    \item Drop the \emph{single lowest} score---your one free card. Average the other five for continuous control (CC), on the /20 scale.
  \end{itemize}
  \[
    \mathrm{CC}=\frac{\text{sum of six scores}-\text{lowest score}}{5}
  \]
  \vspace{-0.8em}
  \[
    \mathrm{Final}=\max\!\left(E,\;\frac{E+\mathrm{CC}}{2}\right)
  \]
  \vspace{0.2em}
  \small
  Example: $0,10,12,14,16,18$ $\rightarrow$ drop $0$, so $\mathrm{CC}=14$.
  If exam mark $E=12$, the final mark is $\max(12,13)=13$.
\end{frame}

\section{Model and simulate}

\begin{frame}{First model: one arrival rate for the whole month}
  Assume a constant rate $r$ VRU entries per minute. For each 15-minute count $X_i$:
  \begin{gather*}
    X_i\sim\operatorname{Poisson}(15r),\qquad
    \mathbb E[X_i]=\operatorname{Var}(X_i)=15r,\\
    \mathbb P(X_i=0)=e^{-15r}.
  \end{gather*}
  \begin{itemize}
    \item The homogeneous Poisson process also assumes independent counts in disjoint intervals.
    \item Fit $r$ from the pooled mean: $\hat r=14.14/15\approx0.943$ entries per minute.
    \item Nights and busy hours are deliberately pooled. This is a baseline to test, not an endorsed staffing model.
  \end{itemize}
\end{frame}

\begin{frame}{The pooled baseline fails two checks}
  \small
  \begin{tabular}{@{}lcc@{}}
    & \textbf{Observed} & \textbf{Constant-rate model}\\[0.35em]
    Mean count & 14.14 & 14.14 (fitted)\\[0.35em]
    Variance of counts & 209.30 & 14.14\\[0.35em]
    Empty-bin fraction & 29.1\% & $e^{-14.14}\approx7.2\times10^{-7}$\\
  \end{tabular}
  \vspace{0.7em}
  \begin{itemize}
    \item The observed variance uses divisor $n$ (Python \texttt{statistics.pvariance}).
    \item The pooled model predicts almost no empty bins; the extract has many.
    \item This challenges a \emph{constant} rate. We must inspect when the empty bins occur.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Simulate arrivals with plain Python}
  \small
\begin{verbatim}
import random, statistics
rate = statistics.mean(observed) / 15  # per minute
rng = random.Random(2026)
simulated = [0] * len(observed)
clock = rng.expovariate(rate)
while clock < 15 * len(observed):
    simulated[int(clock // 15)] += 1
    clock += rng.expovariate(rate)
\end{verbatim}
  \vspace{0.2em}
  Each exponential gap moves the clock to the next arrival. The notebook wraps this in a reusable function with input checks.
\end{frame}

\begin{frame}{Check the simulator; then check the model}
  \begin{itemize}
    \item With a few different seeds, simulated mean counts should stay near 14.14.
    \item The simulated variance/mean ratio should be near 1, not exactly 1 in every run.
    \item Simulated empty-bin share should be near $e^{-14.14}$; the observed share is 29.1\%.
  \end{itemize}
  \vspace{0.6em}
  \textbf{Do not tune the seed to make the simulation resemble the data.} This is a test of the constant-rate assumption, not a search for a lucky run.
\end{frame}

\begin{frame}{Validate by clock hour, then propose a next test}
  \begin{itemize}
    \item The homogeneous model expects about 14.14 entries per 15-minute bin at \emph{every} clock hour.
    \item Observed means: about 0.1 at 03:00 and 32.3 at 10:00, across all November days.
    \item Find where the notebook's simulated hourly profile overpredicts and underpredicts the data.
  \end{itemize}
  \vspace{0.5em}
  \textbf{Next test:} estimate separate rates for comparable time slots and day types; reserve later days to check them. A better hourly profile alone would not prove independent Poisson arrivals.
\end{frame}

\section{Verification and exit check}

\begin{frame}{Use an LLM to check a theoretical claim}
  Ask an LLM:
  \begin{quote}
    ``If counts in periods A and B are each Poisson, is a count from a randomly selected period Poisson with their average rate? Prove or disprove it. Distinguish selection from addition.''
  \end{quote}
  \begin{enumerate}
    \item Use a fair coin to select A or B, with $A\sim\operatorname{Poi}(1)$ and $B\sim\operatorname{Poi}(3)$.
    \item Check the selected count's mean and variance yourself. Compare with $\operatorname{Poi}(2)$.
    \item State separately what happens if independent counts $A$ and $B$ are \emph{added}.
  \end{enumerate}
  \vspace{0.3em}
  \footnotesize Record the LLM claim, its assumptions, your independent check and any correction. Use no raw call records in the prompt.
\end{frame}

\begin{frame}{End-of-session participation check}
  \Large In the final 10 minutes, complete the printed six-question exit quiz handed out in class.\normalsize
  \vspace{0.8em}
  \begin{itemize}
    \item Work individually, without devices or LLMs.
    \item Choose one answer per question.
    \item Give one question about the data or model that you still have.
    \item Hand it in before the group discussion.
  \end{itemize}
  \vspace{0.5em}
  The quiz checks measurement, a basic Python conversion, variability and independent LLM verification.
\end{frame}

\begin{frame}{What we can take forward}
  \begin{itemize}
    \item The bank extract gives a real arrival pattern, with important limits on what it measures.
    \item A simple Poisson model is a useful starting point only when its rate and independence assumptions are stated and checked.
    \item Simulation creates a testable prediction; observed counts show where the homogeneous baseline fails.
    \item A staffing recommendation needs queue, service and abandonment evidence too.
  \end{itemize}
  \vspace{0.5em}
  \footnotesize Sources: \href{https://seelab.net.technion.ac.il/data/}{Technion SEE Lab, AnonymousBank data repository};
  \href{https://doi.org/10.1198/016214504000001808}{Brown et al., ``Statistical Analysis of a Telephone Call Center,'' \emph{JASA} 100 (2005), 36--50}.
\end{frame}
