\iffalse
\begin{frame}{Use an LLM to check a theoretical claim}
  Ask an LLM:
  \begin{quote}
    ``If counts in periods A and B are each Poisson, is a count from a randomly selected period Poisson with their average rate? Prove or disprove it. Distinguish selection from addition.''
  \end{quote}
  \begin{enumerate}
    \item Flip a fair coin: on heads observe one count from period A, on tails one count from period B. You observe one count; you do not add both.
    \item With $A\sim\operatorname{Poi}(1)$ and $B\sim\operatorname{Poi}(3)$, the selected count has $P(X=k)=\tfrac12P(A=k)+\tfrac12P(B=k)$: a mixture.
    \item Check its mean and variance, then state separately what happens if independent counts $A$ and $B$ are \emph{added}.
  \end{enumerate}
  \vspace{0.3em}
  \footnotesize Record the LLM claim, its assumptions, your independent check and any correction. 
\end{frame}
\fi

\begin{frame}{Assessment}
  \begin{itemize}
    \item Graded end-of-session MCQs are scored out of 20. Their number and schedule are to be confirmed. An absence scores zero.
    \item Let $N$ be the number of scheduled quizzes, including any missed. Drop the \emph{single lowest} score (one free card):
  \[
    \mathrm{CC}=\frac{\text{sum of scores}-\text{lowest score}}{N-1}
  \]
  \item Final exam $E$ is scored out of 20. The final mark is the the average of $E$ and CC:
  \[
    \mathrm{Final}=\frac{E+\mathrm{CC}}{2}
  \]
  \end{itemize}
  \small
  Example if $N=6$: $0,10,12,14,16,18$ $\rightarrow$ drop $0$, so $\mathrm{CC}=14$.
  If exam mark $E=12$, the final mark is $mean(12,14)=13$.
\end{frame}

\section{Model and simulate}

\begin{frame}{Poisson process: the basic idea}
  \centering
  \resizebox{0.94\textwidth}{!}{%
  \begin{tikzpicture}[x=0.67cm,y=1cm,font=\small]
    \draw[->,thick] (0,0)--(15.5,0) node[right] {time};
    \draw (0,0.12)--(0,-0.12) node[below] {0};
    \draw (15,0.12)--(15,-0.12) node[below] {15 min};
    \foreach \x in {0.8,2.1,2.8,5.2,7.7,9.1,9.7,12.8,14.2} {
      \fill[AMU_dk2] (\x,0) circle (2.2pt);
    }
    \node[above] at (7.5,0.18) {each dot is one arrival};
  \end{tikzpicture}
  }

  \vspace{0.5em}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.50\textwidth}
      \textbf{Homogeneous-process assumptions}
      \begin{itemize}
        \item Arrivals occur at a constant average rate $r$.
        \item Counts in disjoint intervals are independent.
        \item In a very short interval, two or more arrivals are negligible.
      \end{itemize}
    \end{column}
    \begin{column}{0.47\textwidth}
      \small
      \textbf{Consequences for an interval of length $t$}
      \begin{align*}
        N(t)&\sim\operatorname{Poisson}(rt),\\
        \mathbb E[N(t)]&=\operatorname{Var}(N(t))=rt,\\
        T_{\mathrm{next}}&\sim\operatorname{Exponential}(r),\\
        \mathbb E[T_{\mathrm{next}}]&=1/r.
      \end{align*}
    \end{column}
  \end{columns}

  \vspace{0.3em}
  \small A Poisson \textbf{process} models arrival times. The Poisson \textbf{distribution} describes how many arrivals fall inside a chosen interval.
\end{frame}

\begin{frame}{First baseline: one rate for every interval}
  Deliberately ignore clock time and date. Assume the same constant rate $r$ for all 2,880 intervals in November. For each 15-minute count $X_i$:
  \begin{gather*}
    X_i\sim\operatorname{Poisson}(15r),\qquad
    \mathbb E[X_i]=\operatorname{Var}(X_i)=15r,\\
    \mathbb P(X_i=0)=e^{-15r}.
  \end{gather*}
  \vspace{-0.5cm}
  \begin{itemize}
    \item The homogeneous Poisson process also assumes independent counts in disjoint intervals.
    \item Fit $r$ from the mean of all 2,880 counts: $\hat r=14.14/15\approx0.943$ entries per minute.
    \item This deliberately pools nights, busy hours, weekdays and weekends. We use it as a simple baseline whose failures should reveal which variation matters.
  \end{itemize}
\end{frame}

\begin{frame}{The pooled baseline fails two checks}
  \small
  \begin{tabular}{@{}lcc@{}}
    & \textbf{Observed} & \textbf{Constant-rate model}\\[0.35em]
    Mean count & 14.14 & 14.14 (fitted)\\[0.35em]
    Variance of counts & 209.30 & 14.14\\[0.35em]
    Empty-bin fraction & 29.1\% & $e^{-14.14}\approx7.2\times10^{-7}$\\
  \end{tabular}
  \vspace{0.7em}
  \begin{itemize}
    \item The observed variance uses divisor $n$ (Python \texttt{statistics.pvariance}).
    \item The pooled model predicts almost no empty bins; the extract has many.
    \item This challenges a \emph{constant} rate. We must inspect when the empty bins occur.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Simulate arrivals with plain Python}
  \small
\begin{verbatim}
import random, statistics
rate = statistics.mean(observed) / 15  # per minute
rng = random.Random(2026)
simulated = [0] * len(observed)
clock = rng.expovariate(rate)
while clock < 15 * len(observed):
    simulated[int(clock // 15)] += 1
    clock += rng.expovariate(rate)
\end{verbatim}
  \vspace{0.2em}
  Each exponential gap moves the clock to the next arrival. The notebook wraps this in a reusable function with input checks.
\end{frame}

\begin{frame}{Check the simulator; then check the model}
  \small
  One run of the preceding code: 2,880 intervals, seed \texttt{2026}.
  \vspace{0.5em}

  \begin{tabular}{@{}lrrr@{}}
    & \textbf{Model prediction} & \textbf{Simulated} & \textbf{Observed}\\[0.35em]
    Mean count & 14.14 & 14.25 & 14.14\\[0.35em]
    Variance of counts & 14.14 & 14.52 & 209.30\\[0.35em]
    Empty intervals & $0.000072\%$ & $0\%$ & $29.1\%$\\
  \end{tabular}
  \vspace{0.6em}
  \begin{itemize}
    \item \textbf{Simulator check:} the simulated mean and variance are close to the theoretical values. This supports the implementation; repeat with other seeds.
    \item \textbf{Model check:} the observed counts have much greater variance and many more empty intervals. The constant-rate baseline misses this variation.
  \end{itemize}
  \vspace{0.3em}
  Results vary between runs. Choose the seed before comparing results; do not tune it to match the data. Variances above use divisor $n$.
\end{frame}

\begin{frame}{Validate by clock hour, then propose a next test}
  \begin{itemize}
    \item The homogeneous model expects about 14.14 entries per 15-minute bin at \emph{every} clock hour.
    \item Observed means: about 0.1 at 03:00 and 32.3 at 10:00, across all November days.
    \item Find where the notebook's simulated hourly profile overpredicts and underpredicts the data.
  \end{itemize}
  \vspace{0.5em}
  \textbf{Two sources of variation:} first estimate a rate for each time of day; then check whether those rates also differ between weekdays, weekends or individual dates. Reserve later days to test the fitted rates. A better daily profile alone would not prove independent Poisson arrivals.
\end{frame}
